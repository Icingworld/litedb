## 版本

v0.3.0

## 创建数据库

```sql
create database vec;
```

## 切换数据库

```sql
use vec;
```

## 创建集合

```sql
create collection t_vec (
    id integer,
   	embedding vector(16),
   	type integer,
   	is_effective boolean default true
);
```

```sql
create index idx_id on t_vec(id);
```

## 查询

### l2

```sql
SELECT id, type, l2_distance(embedding, [0.639427, 0.025011, 0.275029, 0.223210, -0.691661, 0.056411, 0.656614, 0.299368, 0.188495, 0.577086, 0.695776, 0.717387, 0.877858, 0.272592, 0.097078, 0.336472])
FROM t_vec
WHERE is_effective = true
ORDER BY l2_distance(embedding, [0.639427, 0.025011, 0.275029, 0.223210, -0.691661, 0.056411, 0.656614, 0.299368, 0.188495, 0.577086, 0.695776, 0.717387, 0.877858, 0.272592, 0.097078, 0.336472]) ASC
LIMIT 5;
```

### cosine

```sql
SELECT id, type, cosine_distance(embedding, [0.639427, 0.025011, 0.275029, 0.223210, -0.691661, 0.056411, 0.656614, 0.299368, 0.188495, 0.577086, 0.695776, 0.717387, 0.877858, 0.272592, 0.097078, 0.336472])
FROM t_vec
ORDER BY cosine_distance(embedding, [0.639427, 0.025011, 0.275029, 0.223210, -0.691661, 0.056411, 0.656614, 0.299368, 0.188495, 0.577086, 0.695776, 0.717387, 0.877858, 0.272592, 0.097078, 0.336472]) ASC
LIMIT 5;
```

### ip

```sql
SELECT id, type, inner_product(embedding, [0.639427, 0.025011, 0.275029, 0.223210, -0.691661, 0.056411, 0.656614, 0.299368, 0.188495, 0.577086, 0.695776, 0.717387, 0.877858, 0.272592, 0.097078, 0.336472])
FROM t_vec
WHERE type = 1
ORDER BY inner_product(embedding, [0.639427, 0.025011, 0.275029, 0.223210, -0.691661, 0.056411, 0.656614, 0.299368, 0.188495, 0.577086, 0.695776, 0.717387, 0.877858, 0.272592, 0.097078, 0.336472]) DESC
LIMIT 5;
```
